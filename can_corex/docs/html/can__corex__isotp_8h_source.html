335 {

343 uint8_t SequenceNumber; /* CF sequence number (0-15) */
344 uint8_t BlockCounter; /* Counter for BS */
345 CCX_TIME_t LastTick; /* Last activity timestamp in base ticks */
346 CCX_HR_TIME_t LastHighResTick; /* Last CF scheduling timestamp for sub-millisecond STmin */
347 uint8_t WaitFramesRemaining; /* Number of WAIT FC frames we can tolerate */
348 CCX_TIME_t STminTicks; /* STmin value from FC converted to base ticks */
349 CCX_HR_TIME_t STminHighResTicks; /* STmin value from FC converted to HR ticks when sub-millisecond */
350 uint8_t STminUsesHighRes; /* 1 when STmin must be enforced in HR domain */
351 uint8_t ActiveTxDL; /* Active link-layer payload for TX */
352 uint8_t MaxWaitFrames; /* Effective FC.WAIT tolerance for this instance */
353 uint8_t InitValid; /* Successful init state */
354 CCX_ISOTP_LengthFormat_t LengthFormat;
355 };
