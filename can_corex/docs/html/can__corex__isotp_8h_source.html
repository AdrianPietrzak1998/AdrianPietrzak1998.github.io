346 CCX_TIME_t STminTicks; /* STmin value from FC converted to base ticks */

348 uint8_t STminUsesHighRes; /* 1 when STmin must be enforced in HR domain */
349 uint8_t ActiveTxDL; /* Active link-layer payload for TX */

351 uint8_t InitValid; /* Successful init state */
